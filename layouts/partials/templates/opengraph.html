{{/* Open Graph meta for huffmanwrites. Override of PaperMod's templates/opengraph.html. Image selection (priority order): 1. .Params.hero_desktop — posts with wired hero images 2. .Params.cover.image — book pages 3. /img/og-default.webp — fallback for home, about, mission, _index pages Twitter Card meta is intentionally NOT emitted (see no-twitter-x memory). Most OG consumers (LinkedIn, Bluesky, Slack, Discord, iMessage, Facebook, Mastodon) read og:* directly — Twitter Card support is the only thing we lose, and that's the explicit intent. */}} {{- with or site.Title site.Params.title | plainify }} {{- end }} {{- $ogTitle := .Title -}} {{- /* The home page's content file (_index.md) is titled "Mission"; link previews should say "Huffman Writes", not "Mission". */ -}} {{- if .IsHome -}} {{- $ogTitle = site.Title -}} {{- end -}} {{- with $ogTitle | plainify }} {{- end }} {{- with or .Description .Summary site.Params.description | plainify | htmlUnescape }} {{- end }} {{- with or .Params.locale site.Language.LanguageCode }} {{- end }} {{- if .IsPage }} {{- with .Section }} {{- end }} {{- $ISO8601 := "2006-01-02T15:04:05-07:00" }} {{- with .PublishDate }} {{- end }} {{- with .Lastmod }} {{- end }} {{- range .GetTerms "tags" | first 6 }} {{- end }} {{- else }} {{- end }} {{/* Image: hero-aware selection. Set og:image, og:image:alt, og:image:width, og:image:height once we've resolved the source. The URL is built with absURL so it follows the site's baseURL (www.huffmanwrites.org in production) — matching the canonical link, since some scrapers don't follow redirects on image URLs. Dimensions are read from the actual image resource; the fallback og-default.webp is 1200x630. */}} {{- $ogImage := "" -}} {{- $ogImageRel := "" -}} {{- $ogImageAlt := "" -}} {{- if .Params.hero_desktop -}} {{- $ogImageRel = .Params.hero_desktop -}} {{- $ogImageAlt = .Params.hero_alt | default "" -}} {{- else if .Params.cover.image -}} {{- $ogImageRel = .Params.cover.image -}} {{- else if .Params.image -}} {{- $ogImageRel = .Params.image -}} {{- end -}} {{- if not $ogImageRel -}} {{- $ogImageRel = "img/og-default.webp" -}} {{- $ogImageAlt = "Huffman Writes - Stoicism, Philosophy, and the Pursuit of a Well-Lived Life" -}} {{- end -}} {{- $ogImage = printf "%s/%s" (strings.TrimSuffix "/" site.BaseURL) (strings.TrimPrefix "/" $ogImageRel) -}} {{- /* Dimensions: read from the actual file. Book covers live in assets/, article heroes and og-default in static/ — probe both with os.Stat, then imageConfig for the dims. */ -}} {{- $ogImagePath := "" -}} {{- range slice (printf "static/%s" $ogImageRel) (printf "assets/%s" $ogImageRel) -}} {{- if and (not $ogImagePath) (fileExists .) -}} {{- $ogImagePath = . -}} {{- end -}} {{- end -}} {{- $ogImageW := 1200 -}} {{- $ogImageH := 630 -}} {{- $ogImageType := "image/webp" -}} {{- if $ogImagePath -}} {{- with imageConfig $ogImagePath -}} {{- $ogImageW = .Width -}} {{- $ogImageH = .Height -}} {{- end -}} {{- $ext := path.Ext $ogImagePath -}} {{- if eq $ext ".jpg" -}}{{- $ogImageType = "image/jpeg" -}} {{- else if eq $ext ".png" -}}{{- $ogImageType = "image/png" -}} {{- end -}} {{- end -}} {{- if and $ogImageRel (not $ogImageAlt) -}} {{- $ogImageAlt = .Title -}} {{- end -}} {{- with $ogImageAlt }} {{- end }} {{- with .Params.audio }} {{- range . | first 6 }} {{- end }} {{- end }} {{- with .Params.videos }} {{- range . | first 6 }} {{- end }} {{- end }} {{- range .GetTerms "series" }} {{- range .Pages | first 7 }} {{- if ne $ . }} {{- end }} {{- end }} {{- end }} {{- with site.Params.social }} {{- if reflect.IsMap . }} {{- with .facebook_app_id }} {{- else }} {{- with .facebook_admin }} {{- end }} {{- end }} {{- end }} {{- end }} {{- with (.Param "social.fediverse_creator") }} {{- end }}
