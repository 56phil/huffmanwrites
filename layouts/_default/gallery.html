{{ define "main" }} {{- /* Newest-first. gallery.yml is maintained in insertion order, so the raw list is oldest-first and the most recent hero art would otherwise land on the last page, where nobody scrolls. Reversing here rather than reordering the file keeps the data appendable: new entries still go on the end, and the display order stays correct without touching 100+ existing records. Pagination indexes into this slice, so every page flips consistently. */ -}} {{- $galleryItems := collections.Reverse hugo.Data.gallery -}} {{- $itemsPerPage := 12 -}} {{- $totalItems := len $galleryItems -}} {{- $totalPages := add (div (sub $totalItems 1) $itemsPerPage) 1 | int -}} {{- $baseURL := (site.GetPage "/gallery").RelPermalink -}} {{- /* Determine current page from URL path */ -}} {{- $currentPage := 1 -}} {{- $pathParts := split (strings.TrimSuffix "/" .RelPermalink) "/" -}} {{- $lastPart := index $pathParts (sub (len $pathParts) 1) -}} {{- $secondLast := "" -}} {{- if gt (len $pathParts) 2 -}} {{- $secondLast = index $pathParts (sub (len $pathParts) 2) -}} {{- end -}} {{- if and (eq $secondLast "page") (gt (len $pathParts) 2) -}} {{- $currentPage = int $lastPart -}} {{- end -}} {{- $offset := mul (sub $currentPage 1) $itemsPerPage -}} {{- $pageEnd := add $offset $itemsPerPage -}} {{- /* All items render on every page so the lightbox can navigate the full gallery; cards outside the current page's range are hidden with CSS. DOM order is global, so GLightbox's prev/next walks all 80 images regardless of which page you start on. */ -}}

"Date" "desc") 0 -}} {{- $linkRaw = $p.RelPermalink -}} {{- with $p.Params.hero_desktop -}} {{- $imgRaw = printf "/%s" . -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Guard the empty case: `site.GetPage ""` returns the HOME page, not nil, so an unresolved card would render a "Read Post" pointing at "/" — a link that works and goes somewhere wrong, which is worse than no link at all. */ -}} {{- $linkedPage := "" -}} {{- with $linkRaw -}} {{- $linkedPage = site.GetPage . -}} {{- end -}} {{- $caption := $item.caption -}} {{- if $linkedPage -}} {{- if $linkedPage.Description -}} {{- $caption = $linkedPage.Description -}} {{- else if $linkedPage.Params.description -}} {{- $caption = $linkedPage.Params.description -}} {{- end -}} {{- end -}} {{- $hidden := "" -}} {{- if or (lt $i $offset) (ge $i $pageEnd) -}} {{- $hidden = " hidden" -}} {{- end -}}
